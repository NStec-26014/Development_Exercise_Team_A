-- 顧客 (customer)
CREATE TABLE customer (
    id SERIAL PRIMARY KEY,
    name VARCHAR(20),
    name_kana VARCHAR(20),
    address1 VARCHAR(100),
    address2 VARCHAR(100),
    phone_number VARCHAR(20),
    mail_address VARCHAR(200),
    username VARCHAR(30),
    password VARCHAR(200),
    register_date TIMESTAMP
);

-- 部署 (department)
CREATE TABLE department (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)
);

-- 商品カテゴリ (product_category)
CREATE TABLE product_category (
    id SERIAL PRIMARY KEY,
    name VARCHAR(20)
);

-- 注文ステータス (order_status)
CREATE TABLE order_status (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)
);

-- 支払い方法 (payment_method)
CREATE TABLE payment_method (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)
);


-- 社員 (employee)
CREATE TABLE employee (
    id SERIAL PRIMARY KEY,
    department_id INTEGER REFERENCES department(id),
    name VARCHAR(100),
    name_kana VARCHAR(100)
);

-- 社員アカウント (employee_account)
CREATE TABLE employee_account (
    id SERIAL PRIMARY KEY,
    employee_id INTEGER REFERENCES employee(id),
    name VARCHAR(20),
    password VARCHAR(200)
);

-- 商品 (product)
CREATE TABLE product (
    id SERIAL PRIMARY KEY,
    product_category_id INTEGER REFERENCES product_category(id),
    name VARCHAR(20),
    price INTEGER,
    image_url VARCHAR(200),
    delete_flag INTEGER
);

-- 商品在庫 (product_stock)
CREATE TABLE product_stock (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES product(id),
    quantity INTEGER
);

-- 注文 (orders)
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    customer_id INTEGER REFERENCES customer(id),
    order_status_id INTEGER REFERENCES order_status(id),
    payment_method_id INTEGER REFERENCES payment_method(id),
    order_date TIMESTAMP,
    amount_total INTEGER
);

-- 注文明細 (order_detail)
CREATE TABLE order_detail (
    id SERIAL PRIMARY KEY,
    order_id INTEGER REFERENCES orders(id),
    product_id INTEGER REFERENCES product(id),
    customer_id INTEGER REFERENCES customer(id),
    count INTEGER
);
